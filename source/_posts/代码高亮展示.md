## JavaScript

```javascript
// 斐波那契数列(记忆化)
const memo = new Map();

function fib(n) {
  if (n <= 1) return n;
  if (memo.has(n)) return memo.get(n);
  const result = fib(n - 1) + fib(n - 2);
  memo.set(n, result);
  return result;
}

console.log(fib(40)); // 102334155
```

## Python

```python
import asyncio

async def fetch_data(url: str) -> dict:
    """模拟异步请求"""
    await asyncio.sleep(1)
    return {"url": url, "status": 200}

async def main():
    results = await asyncio.gather(
        fetch_data("https://api.example.com/a"),
        fetch_data("https://api.example.com/b"),
    )
    for r in results:
        print(r["url"], r["status"])

asyncio.run(main())
```

## Bash

```bash
#!/bin/bash
# 统计当前目录下各类型文件数量
for ext in $(ls | awk -F. 'NF>1{print $NF}' | sort -u); do
  count=$(ls *."$ext" 2>/dev/null | wc -l)
  echo "$ext: $count 个文件"
done
```

SELECT
  p.id,
  p.title,
  COUNT(c.id) AS comment_count
FROM posts p
LEFT JOIN comments c ON c.post_id = p.id
WHERE p.created_at >= '2026-01-01'
GROUP BY p.id, p.title
ORDER BY comment_count DESC
LIMIT 10;
